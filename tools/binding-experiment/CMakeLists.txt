cmake_minimum_required(VERSION 3.30)
project(pocket_binding_experiment LANGUAGES C CXX)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(POCKET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(POCKET_LUABRIDGE2_SOURCE "${POCKET_ROOT}/build/binding-baseline-phase11/include" CACHE PATH "Retained LuaBridge 2.7 include root")
if(NOT EXISTS "${POCKET_LUABRIDGE2_SOURCE}/LuaBridge/LuaBridge.h")
    message(FATAL_ERROR "Set POCKET_LUABRIDGE2_SOURCE to retained 2.7 headers; production now uses LuaBridge3.")
endif()
set(POCKET_LUABRIDGE3_SOURCE "" CACHE PATH "Pinned LuaBridge3 checkout's Source directory")
if(NOT EXISTS "${POCKET_LUABRIDGE3_SOURCE}/LuaBridge/LuaBridge.h")
    message(FATAL_ERROR "Set POCKET_LUABRIDGE3_SOURCE to the pinned candidate's Source directory; no automatic download occurs.")
endif()
option(POCKET_BINDING_SANITIZE "Instrument the isolated Lua library and probes with ASan/UBSan" OFF)
if(POCKET_BINDING_SANITIZE)
    if(MSVC)
        message(FATAL_ERROR "This probe's combined ASan/UBSan configuration requires Clang or GCC.")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${POCKET_ROOT}/thirdparty/Lua/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/lua[c]?\\.c$")
add_library(probe_lua STATIC ${LUA_SOURCES})
target_include_directories(probe_lua PUBLIC "${POCKET_ROOT}/thirdparty/Lua")
if(UNIX)
    target_link_libraries(probe_lua PUBLIC m ${CMAKE_DL_LIBS})
endif()
add_library(probe_budget STATIC "${POCKET_ROOT}/src/engine/scripting/ScriptBudget.cpp")
target_include_directories(probe_budget PUBLIC "${POCKET_ROOT}/include/engine")
target_link_libraries(probe_budget PUBLIC probe_lua)

include(CTest)
foreach(variant IN ITEMS baseline candidate)
    foreach(probe IN ITEMS semantics budget_errors benchmark)
        set(target "binding_${probe}_${variant}")
        add_executable(${target} "${probe}.cpp")
        target_link_libraries(${target} PRIVATE probe_budget)
        if(variant STREQUAL "candidate")
            target_compile_definitions(${target} PRIVATE POCKET_BINDING_CANDIDATE=1 LUABRIDGE_SAFE_LUA_C_EXCEPTION_HANDLING=1)
            target_include_directories(${target} SYSTEM PRIVATE "${POCKET_LUABRIDGE3_SOURCE}")
        else()
            target_compile_definitions(${target} PRIVATE POCKET_BINDING_CANDIDATE=0)
            target_include_directories(${target} SYSTEM PRIVATE "${POCKET_LUABRIDGE2_SOURCE}")
        endif()
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        if(NOT probe STREQUAL "benchmark")
            add_test(NAME ${target} COMMAND ${target})
            set_tests_properties(${target} PROPERTIES TIMEOUT 30)
        endif()
    endforeach()
endforeach()
